// 目的地:從遠處朝地標直線慢慢滑行,到附近停住懸停(不繞圈,比較不會暈) // r = 懸停時離地標的距離、alt = 飛行高度、top = 看向的高度(公尺) var PLACES = [ { name: '台北101', lat: 25.0340, lng: 121.5645, r: 900, alt: 650, top: 300 }, { name: '京都 清水寺', lat: 34.9949, lng: 135.7850, r: 600, alt: 420, top: 120 }, { name: '日月潭', lat: 23.8572, lng: 120.9150, r: 2500, alt: 1400, top: 760 }, { name: '富士山', lat: 35.3606, lng: 138.7274, r: 9000, alt: 5200, top: 3300 }, { name: '巴黎 鐵塔', lat: 48.8584, lng: 2.2945, r: 800, alt: 450, top: 180 }, { name: '紐約 曼哈頓', lat: 40.7580, lng: -73.9855, r: 1200, alt: 550, top: 200 } ]; var GLIDE_SEC = 240; // 速度 1 倍時,一段滑行約 4 分鐘(約原本繞圈速度的 1/3) var START_FAR = 3; // 從 3 倍懸停距離的地方出發 var key = '', map = null, place = null; var leg = 0; // 第幾個方向接近(每次「換角度」轉 90 度) var progress = 0; // 0 = 出發點,1 = 懸停點 var paused = false, last = 0; var speed = 1, altScale = 1; var lookYaw = 0, lookPitch = 0; // 轉頭角度(度):向右、向上為正,由 App 傳入 function $(id) { return document.getElementById(id); } function msg(html) { $('msg').innerHTML = html; } // App 傳入金鑰(只在 App 內建頁面才會傳) window.vrSetKey = function (k) { key = k || ''; if (!key) { msg('還沒有設定 Google 金鑰
'); return; } msg(''); }; // App 傳入頭的方向、飛行設定、指令 window.vrLook = function (yaw, pitch) { lookYaw = yaw; lookPitch = pitch; }; window.vrFlightConfig = function (s, a) { speed = s; altScale = a; }; window.vrFlight = function (cmd) { if (!place) return false; if (cmd === 'pause') paused = !paused; else if (cmd === 'restart') fadeTo(function () { progress = 0; paused = false; }); else if (cmd === 'next') fadeTo(function () { leg++; progress = 0; paused = false; }); return true; }; // 面板顯示用的狀態 window.vrFlightState = function () { return place ? { name: place.name, paused: paused, arrived: progress >= 1 } : null; }; // 返回鍵:正在飛就回目的地選單;已在選單回傳 false(App 回首頁) window.vrBack = function () { if (!place) return false; place = null; showMenu(); return true; }; window.gm_authFailure = function () { showMenu(); msg('金鑰無效,或沒有啟用 Maps JavaScript API
'); }; function showMenu() { $('menu').style.display = 'flex'; } // 換場景時先變黑再變亮,避免畫面突然跳動造成頭暈 function fadeTo(change) { var f = $('fade'); f.style.opacity = 1; setTimeout(function () { change(); f.style.opacity = 0; }, 650); } PLACES.forEach(function (p, i) { var b = document.createElement('button'); b.textContent = p.name; b.onclick = function () { go(i); }; $('list').appendChild(b); }); var loading = null; function loadMaps() { if (loading) return loading; loading = new Promise(function (ok, fail) { window.__mapsReady = ok; var s = document.createElement('script'); s.src = 'https://maps.googleapis.com/maps/api/js?key=' + encodeURIComponent(key) + '&v=beta&libraries=maps3d&language=zh-TW&callback=__mapsReady'; s.onerror = function () { fail(new Error('無法連線到 Google 地圖')); }; document.head.appendChild(s); }); return loading; } function go(i) { if (!key) { window.vrSetKey(''); return; } place = PLACES[i]; leg = 0; progress = 0; paused = false; msg('載入中…'); loadMaps().then(function () { return google.maps.importLibrary('maps3d'); }).then(function (lib) { if (!map) { var opts = { center: { lat: place.lat, lng: place.lng, altitude: place.alt }, range: 1, tilt: 70, heading: 0 }; try { map = new lib.Map3DElement(Object.assign({ mode: 'SATELLITE', defaultUIHidden: true }, opts)); } catch (e) { map = new lib.Map3DElement(opts); } $('map').appendChild(map); } msg(''); $('menu').style.display = 'none'; var t = $('title'); t.textContent = place.name; t.style.opacity = 1; setTimeout(function () { t.style.opacity = 0; }, 4000); last = performance.now(); requestAnimationFrame(fly); }).catch(function (e) { msg('無法載入 3D 地圖:' + (e && e.message ? e.message : e)); }); } var M_PER_DEG = 111320; function ease(x) { return x * x * (3 - 2 * x); } // 起步、停下都慢慢來 function fly(now) { if (!place || $('menu').style.display !== 'none') return; requestAnimationFrame(fly); var dt = Math.min(0.1, (now - last) / 1000); last = now; if (!paused && progress < 1) progress = Math.min(1, progress + dt * speed / GLIDE_SEC); var p = place; var bearing = (180 + leg * 90) * Math.PI / 180; // 出發方向(從地標看出去):先從南邊接近 var dist = p.r * (START_FAR - (START_FAR - 1) * ease(progress)); var east = Math.sin(bearing) * dist, north = Math.cos(bearing) * dist; var lat = p.lat + north / M_PER_DEG; var lng = p.lng + east / (M_PER_DEG * Math.cos(p.lat * Math.PI / 180)); var alt = p.top + (p.alt - p.top) * altScale; var toCenter = (Math.atan2(-east, -north) * 180 / Math.PI + 360) % 360; // 看向地標的方位(滑行中不變) var down = Math.atan2(alt - p.top, dist) * 180 / Math.PI; // 看向地標要低頭幾度 map.center = { lat: lat, lng: lng, altitude: alt }; map.range = 1; map.heading = (toCenter + lookYaw + 360) % 360; map.tilt = Math.max(0, Math.min(90, 90 - down + lookPitch)); // 90 = 看水平 } // 2D 時點兩下回目的地選單 document.addEventListener('dblclick', function () { window.vrBack(); }); 目的地選單(App 的返回鍵會先回上一頁;這裡提供點兩下回選單) document.addEventListener('dblclick', function () { if (place) { place = null; showMenu(); } });
